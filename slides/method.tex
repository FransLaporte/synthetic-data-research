\begin{frame}{Proposed review method}
\begin{enumerate}
\item \textbf{Search}: database queries and citation tracing.
\item \textbf{Screen}: independent decisions, then reconciliation.
\item \textbf{Extract}: one study record, checked by the other author.
\item \textbf{Synthesise}: compare compatible tasks and protocols.
\end{enumerate}
\vspace{4mm}
\begin{block}{Evidence trail}
Record exact searches, exclusion reasons, source locations and limitations.
\end{block}
\end{frame}
